! ==============================================================================
! 5. SWAC3 (GES1 - VLAN 30)
! ==============================================================================
enable
configure terminal
hostname SWAC3
no ip domain-lookup
ip domain-name itla.edu.do
enable secret class
username admin privilege 15 secret class
service password-encryption
banner motd # Acceso restringido - ITLA TI-203 Equipo 2 #
! ACL estandar 10: solo la red de Gestion puede entrar por VTY
access-list 10 permit 172.18.10.64 0.0.0.63
line console 0
 password class
 login
 logging synchronous
 exec-timeout 0 0
line vty 0 4
 access-class 10 in
 login local
 transport input ssh
 exec-timeout 0 0
exit
crypto key generate rsa modulus 1024
ip ssh version 2
vtp domain ITLA
vtp mode client
vtp password class
spanning-tree mode rapid-pvst
interface range ethernet 0/0 - 1
 description Trunks hacia SWM1 y SWM2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
interface ethernet 3/0
 description GES1
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
! Gestion del switch (VLAN 30)
interface vlan 30
 ip address 172.18.10.73 255.255.255.192
 no shutdown
ip default-gateway 172.18.10.65
end
write memory
