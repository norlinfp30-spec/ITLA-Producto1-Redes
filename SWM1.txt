! ==============================================================================
! 1. SWM1 (Switch Multicapa - Gateway Principal) - ID OSPF 2713
! ==============================================================================
enable
configure terminal
hostname SWM1
no ip domain-lookup
ip domain-name itla.edu.do
enable secret class
username admin privilege 15 secret class
service password-encryption
banner motd # Acceso restringido - ITLA TI-203 Equipo 2 #
! ACL estandar 10: solo la red de Gestion puede entrar por VTY
access-list 10 permit 172.18.10.64 0.0.0.63
line console 0
 password class
 login
 logging synchronous
 exec-timeout 0 0
line vty 0 4
 access-class 10 in
 login local
 transport input ssh
 exec-timeout 0 0
exit
crypto key generate rsa modulus 1024
ip ssh version 2
ip routing
ipv6 unicast-routing

! VTP Server y VLANs
vtp domain ITLA
vtp mode server
vtp password class
vlan 10
 name Estudiantes
vlan 20
 name Docentes
vlan 30
 name Gestion
vlan 40
 name Voz
vlan 99
 name Blackhull
exit

! Spanning-Tree (SWM1 root VLAN 10,30,99 / SWM2 root 20,40)

spanning-tree vlan 10,30,99 priority 4096
spanning-tree vlan 20,40 priority 8192

! Enlace enrutado hacia R1-Core

 description Hacia R1-Core
 no switchport
 ip address 172.18.10.194 255.255.255.252
 ipv6 address 2001:25:18:9::2/64
 ip ospf network point-to-point
 ipv6 ospf network point-to-point
 no shutdown

! EtherChannel LACP hacia SWM2 (primero el Port-channel)
interface port-channel 1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,99
exit
interface range ethernet 0/1 - 3
 channel-protocol lacp
 channel-group 1 mode active
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,99
 no shutdown
exit

! Trunks hacia SWAC1, SWAC2 y SWAC3
interface range ethernet 1/0 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
exit
